local function getLocalCharacter()
    local char = LocalPlayer.Character
    if not char then return nil, nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 then return nil, nil end
    return char, hrp
end

local function isAlive(plr)
    if not plr or not plr.Character then return false end
    local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
    local hum = plr.Character:FindFirstChildOfClass("Humanoid")
    return hrp ~= nil and hum ~= nil and hum.Health > 0
end

local function setTarget(plr)
    if plr == nil then
        targetPlayerName = ""; targetPlayerPtr = nil
    else
        targetPlayerName = plr.Name; targetPlayerPtr = plr
    end
end

local function getTargetHRP()
    local _, localHRP = getLocalCharacter()
    if not localHRP then return nil, nil end

    if targetPlayerPtr ~= nil or targetPlayerName ~= "" then
        local plr = targetPlayerPtr
        if plr and not plr.Parent then setTarget(nil); plr = nil end
        if not plr and targetPlayerName ~= "" then
            plr = Players:FindFirstChild(targetPlayerName)
        end
        if plr and plr ~= LocalPlayer then
            if isAlive(plr) then
                return plr, plr.Character:FindFirstChild("HumanoidRootPart")
            else
                return nil, nil
            end
        end
        setTarget(nil)
    end

    local bestPlr, bestHRP, bestDist = nil, nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        if not isAlive(plr) then continue end
        local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
        if not hrp then continue end
        local d = (localHRP.Position - hrp.Position).Magnitude
        if d < bestDist then bestDist = d; bestPlr = plr; bestHRP = hrp end
    end
    return bestPlr, bestHRP
end

local function getInFrontCFrame(targetHRP)
    local inFront = targetHRP.CFrame * CFrame.new(0, 0, -OFFSET_DISTANCE)
    return CFrame.lookAt(inFront.Position, targetHRP.Position)
end

local function teleportInFront()
    local _, localHRP = getLocalCharacter()
    if not localHRP then return end
    local _, targetHRP = getTargetHRP()
    if not targetHRP then return end
    localHRP.CFrame = getInFrontCFrame(targetHRP)
end

-- ============== AUTO RESPAWN ==============
-- Pure remote fire only — no input simulation.
-- RunService.Heartbeat ticks regardless of OS window focus,
-- so tabbing out never breaks the respawn loop.

local function tryRespawnRemote()
    local rs    = game:GetService("ReplicatedStorage")
    local duels = rs:FindFirstChild("Remotes")
        and rs.Remotes:FindFirstChild("Duels")
    local remote = duels and duels:FindFirstChild("RespawnNow")
    if remote and remote:IsA("RemoteEvent") then
        pcall(function() remote:FireServer() end)
        return true
    end
    return false
end

local function watchHumanoid(char)
    if respawnWatching then return end
    local hum = char:WaitForChild("Humanoid", 5)
    if not hum then return end
    respawnWatching = true

    hum.Died:Connect(function()
        if not AUTO_RESPAWN then respawnWatching = false; return end
        task.spawn(function()
            local deadChar = char
            local attempts = 0
            -- Heartbeat-based — engine-level, focus-independent
            local conn
            conn = RunService.Heartbeat:Connect(function()
                if not AUTO_RESPAWN
                or (LocalPlayer.Character ~= deadChar and LocalPlayer.Character ~= nil)
                or attempts >= 15 then
                    conn:Disconnect()
                    respawnWatching = false
                end
            end)
            -- fire remote every RESPAWN_DELAY seconds
            while respawnWatching and attempts < 15 and AUTO_RESPAWN do
                task.wait(RESPAWN_DELAY)
                if LocalPlayer.Character ~= deadChar
                   and LocalPlayer.Character ~= nil then
                    break
                end
                tryRespawnRemote()
                attempts += 1
            end
            conn:Disconnect()
            respawnWatching = false
        end)
    end)
end

local function hookCharacter(char)
    if not char then return end
    respawnWatching = false
    task.delay(0.1, function() watchHumanoid(char) end)
end

LocalPlayer.CharacterAdded:Connect(hookCharacter)
if LocalPlayer.Character then hookCharacter(LocalPlayer.Character) end

local function makePlayerBtn(plr)
    local isNearest = (plr == nil)
    local displayName = isNearest
        and "⟳ Nearest (auto)"
        or  plr.DisplayName .. "  [" .. plr.Name .. "]"
    local btn = Instance.new("TextButton", scrollFrame)
    btn.Size = UDim2.new(1,-8,0,24)
    btn.BackgroundColor3 = Color3.fromRGB(35,35,55)
    btn.Text = displayName
    btn.TextColor3 = isNearest and Color3.fromRGB(100,220,100) or Color3.fromRGB(220,220,255)
    btn.TextSize = 11; btn.Font = Enum.Font.Gotham
    btn.TextXAlignment = Enum.TextXAlignment.Left; btn.BorderSizePixel = 0
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0,5)
    local pad = Instance.new("UIPadding", btn); pad.PaddingLeft = UDim.new(0,8)

    if isNearest and targetPlayerPtr == nil and targetPlayerName == "" then
        btn.BackgroundColor3 = Color3.fromRGB(60,60,120); selectedBtn = btn
    elseif not isNearest and plr and plr.Name == targetPlayerName then
        btn.BackgroundColor3 = Color3.fromRGB(60,60,120); selectedBtn = btn
    end

    btn.MouseButton1Click:Connect(function()
        if selectedBtn then selectedBtn.BackgroundColor3 = Color3.fromRGB(35,35,55) end
        btn.BackgroundColor3 = Color3.fromRGB(60,60,120); selectedBtn = btn
        if isNearest then
            setTarget(nil); targetDisplay.Text = "Target: Nearest"
        else
            setTarget(plr); targetDisplay.Text = "Target: " .. plr.DisplayName
        end
    end)
    return btn
end

local function rebuildPlayerList()
    for _, c in ipairs(scrollFrame:GetChildren()) do
        if c:IsA("TextButton") then c:Destroy() end
    end
    selectedBtn = nil; makePlayerBtn(nil)
    local count = 1
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then makePlayerBtn(plr); count += 1 end
    end
    scrollFrame.CanvasSize = UDim2.new(0,0,0, count * 26)
end

local function updateStatus()
    if followEnabled then
        local plr, hrp = getTargetHRP()
        if hrp then
            StatusLabel.Text = "Following: " .. (plr and plr.DisplayName or "?")
            StatusLabel.TextColor3 = Color3.fromRGB(100,200,255)
        else
            StatusLabel.Text = "Follow ON — no valid target"
            StatusLabel.TextColor3 = Color3.fromRGB(255,200,100)
        end
    elseif orbitEnabled then
        StatusLabel.Text = "Orbit ON — keybind to TP"
        StatusLabel.TextColor3 = Color3.fromRGB(200,200,100)
    else
        StatusLabel.Text = "Status: Idle"
        StatusLabel.TextColor3 = Color3.fromRGB(150,255,150)
    end
end

local function stopFollow()
    if followConnection then followConnection:Disconnect(); followConnection = nil end
    accumDT = 0
end

local function startFollow()
    stopFollow()
    followConnection = RunService.Heartbeat:Connect(function(dt)
        if not followEnabled then stopFollow(); return end
        accumDT += dt
        if accumDT < FOLLOW_UPDATE_RATE then return end
        accumDT = 0
        local _, localHRP = getLocalCharacter()
        if not localHRP then return end
        local _, targetHRP = getTargetHRP()
        if not targetHRP then return end
        localHRP.CFrame = getInFrontCFrame(targetHRP)
        updateStatus()
    end)
end

print("  Respawn  : remote fire, focus-independent") -- ============== AUTO RESPAWN ==============
-- Fires RespawnNow remote immediately on death, then every
-- RESPAWN_INTERVAL seconds via Heartbeat accumulator.
-- No task.wait in the fire path — no drift, no focus dependency.

local RESPAWN_INTERVAL = 0.2   -- how often to retry after first fire

local function tryRespawnRemote()
    local rs     = game:GetService("ReplicatedStorage")
    local duels  = rs:FindFirstChild("Remotes") and rs.Remotes:FindFirstChild("Duels")
    local remote = duels and duels:FindFirstChild("RespawnNow")
    if remote and remote:IsA("RemoteEvent") then
        pcall(function() remote:FireServer() end)
        return true
    end
    return false
end

local function watchHumanoid(char)
    if respawnWatching then return end
    local hum = char:WaitForChild("Humanoid", 5)
    if not hum then return end
    respawnWatching = true

    hum.Died:Connect(function()
        if not AUTO_RESPAWN then respawnWatching = false; return end
        task.spawn(function()
            local deadChar   = char
            local accum      = 0
            local attempts   = 0
            local maxAttempts = 20   -- 20 × 0.2s = 4s max window

            -- fire immediately on death — no initial delay
            tryRespawnRemote()
            attempts += 1

            local conn
            conn = RunService.Heartbeat:Connect(function(dt)
                -- already respawned — clean up
                if LocalPlayer.Character ~= deadChar
                   and LocalPlayer.Character ~= nil then
                    conn:Disconnect()
                    respawnWatching = false
                    return
                end

                -- toggle turned off mid-respawn
                if not AUTO_RESPAWN then
                    conn:Disconnect()
                    respawnWatching = false
                    return
                end

                -- hit attempt cap
                if attempts >= maxAttempts then
                    conn:Disconnect()
                    respawnWatching = false
                    return
                end

                -- accumulate dt, fire every RESPAWN_INTERVAL
                accum += dt
                if accum >= RESPAWN_INTERVAL then
                    accum = 0
                    tryRespawnRemote()
                    attempts += 1
                end
            end)
        end)
    end)
end

local function hookCharacter(char)
    if not char then return end
    respawnWatching = false
    task.delay(0.1, function() watchHumanoid(char) end)
end

LocalPlayer.CharacterAdded:Connect(hookCharacter)
if LocalPlayer.Character then hookCharacter(LocalPlayer.Character) end
